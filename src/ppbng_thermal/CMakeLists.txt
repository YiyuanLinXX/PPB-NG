cmake_minimum_required(VERSION 3.16)
project(ppbng_thermal LANGUAGES CXX)
find_package(ament_cmake REQUIRED)

option(PPBNG_THERMAL_ENABLE_SPINNAKER "Build the real A6701 Spinnaker backend" OFF)
set(SPINNAKER_ROOT "C:/Program Files/Teledyne/Spinnaker" CACHE PATH "Spinnaker SDK root")

add_library(${PROJECT_NAME} src/thermal_adapter.cpp src/mock_thermal_backend.cpp src/runtime_contract.cpp src/sequence_association.cpp src/recovery_supervisor.cpp)
if(PPBNG_THERMAL_ENABLE_SPINNAKER)
  find_path(SPINNAKER_INCLUDE_DIR Spinnaker.h PATHS "${SPINNAKER_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
  find_library(SPINNAKER_LIBRARY NAMES Spinnaker_v140 PATHS "${SPINNAKER_ROOT}/lib64/vs2015" NO_DEFAULT_PATH REQUIRED)
  target_sources(${PROJECT_NAME} PRIVATE src/spinnaker_a6701_backend.cpp)
  target_include_directories(${PROJECT_NAME} PRIVATE "${SPINNAKER_INCLUDE_DIR}")
  target_link_libraries(${PROJECT_NAME} PRIVATE "${SPINNAKER_LIBRARY}")
  target_compile_definitions(${PROJECT_NAME} PUBLIC PPBNG_THERMAL_WITH_SPINNAKER=1)

  # Read-only hardware inventory utility. It deliberately uses only the
  # transport-layer device map and never initializes or configures a camera.
  add_executable(ppbng_spinnaker_inventory src/spinnaker_inventory.cpp)
  target_compile_features(ppbng_spinnaker_inventory PRIVATE cxx_std_17)
  target_include_directories(ppbng_spinnaker_inventory PRIVATE "${SPINNAKER_INCLUDE_DIR}")
  target_link_libraries(ppbng_spinnaker_inventory PRIVATE "${SPINNAKER_LIBRARY}")
  add_executable(ppbng_a6701_readonly_probe src/a6701_readonly_probe.cpp)
  target_compile_features(ppbng_a6701_readonly_probe PRIVATE cxx_std_17)
  target_include_directories(ppbng_a6701_readonly_probe PRIVATE "${SPINNAKER_INCLUDE_DIR}")
  target_link_libraries(ppbng_a6701_readonly_probe PRIVATE "${SPINNAKER_LIBRARY}")
  # Minimal allow-listed health check. Kept as a separate executable so an
  # obsolete diagnostic process cannot lock the corrected binary on Windows.
  add_executable(ppbng_a6701_health_check src/a6701_readonly_probe.cpp)
  target_compile_features(ppbng_a6701_health_check PRIVATE cxx_std_17)
  target_include_directories(ppbng_a6701_health_check PRIVATE "${SPINNAKER_INCLUDE_DIR}")
  target_link_libraries(ppbng_a6701_health_check PRIVATE "${SPINNAKER_LIBRARY}")
  add_executable(ppbng_a6701_acquisition_probe src/a6701_acquisition_probe.cpp)
  target_compile_features(ppbng_a6701_acquisition_probe PRIVATE cxx_std_17)
  target_include_directories(ppbng_a6701_acquisition_probe PRIVATE "${SPINNAKER_INCLUDE_DIR}")
  target_link_libraries(ppbng_a6701_acquisition_probe PRIVATE "${SPINNAKER_LIBRARY}")
  add_executable(ppbng_a6701_external_capture src/a6701_external_capture.cpp)
  target_compile_features(ppbng_a6701_external_capture PRIVATE cxx_std_17)
  target_include_directories(ppbng_a6701_external_capture PRIVATE "${SPINNAKER_INCLUDE_DIR}")
  target_link_libraries(ppbng_a6701_external_capture PRIVATE "${SPINNAKER_LIBRARY}")
  add_executable(ppbng_a6701_software_reset src/a6701_software_reset.cpp)
  target_compile_features(ppbng_a6701_software_reset PRIVATE cxx_std_17)
  target_include_directories(ppbng_a6701_software_reset PRIVATE "${SPINNAKER_INCLUDE_DIR}")
  target_link_libraries(ppbng_a6701_software_reset PRIVATE "${SPINNAKER_LIBRARY}")
  add_executable(ppbng_a6701_recover_idle src/a6701_recover_idle.cpp)
  target_compile_features(ppbng_a6701_recover_idle PRIVATE cxx_std_17)
  target_include_directories(ppbng_a6701_recover_idle PRIVATE "${SPINNAKER_INCLUDE_DIR}")
  target_link_libraries(ppbng_a6701_recover_idle PRIVATE "${SPINNAKER_LIBRARY}")
  add_executable(ppbng_a6701_focus_recorder src/a6701_focus_recorder.cpp)
  target_compile_features(ppbng_a6701_focus_recorder PRIVATE cxx_std_17)
  target_include_directories(ppbng_a6701_focus_recorder PRIVATE "${SPINNAKER_INCLUDE_DIR}")
  target_link_libraries(ppbng_a6701_focus_recorder PRIVATE "${SPINNAKER_LIBRARY}")
  add_executable(ppbng_a6701_uno_r4_test_node src/a6701_uno_r4_test_node.cpp)
  target_compile_features(ppbng_a6701_uno_r4_test_node PRIVATE cxx_std_17)
  target_include_directories(ppbng_a6701_uno_r4_test_node PRIVATE "${SPINNAKER_INCLUDE_DIR}")
  target_link_libraries(ppbng_a6701_uno_r4_test_node "${SPINNAKER_LIBRARY}")
  if(MSVC)
    target_compile_options(ppbng_spinnaker_inventory PRIVATE /W4 /permissive-)
    target_compile_options(ppbng_a6701_readonly_probe PRIVATE /W4 /permissive-)
    target_compile_options(ppbng_a6701_health_check PRIVATE /W4 /permissive-)
    target_compile_options(ppbng_a6701_acquisition_probe PRIVATE /W4 /permissive-)
    target_compile_options(ppbng_a6701_external_capture PRIVATE /W4 /permissive-)
    target_compile_options(ppbng_a6701_software_reset PRIVATE /W4 /permissive-)
    target_compile_options(ppbng_a6701_recover_idle PRIVATE /W4 /permissive-)
    target_compile_options(ppbng_a6701_focus_recorder PRIVATE /W4 /permissive-)
    target_compile_options(ppbng_a6701_uno_r4_test_node PRIVATE /W4 /permissive-)
  else()
    target_compile_options(ppbng_spinnaker_inventory PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(ppbng_a6701_readonly_probe PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(ppbng_a6701_health_check PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(ppbng_a6701_acquisition_probe PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(ppbng_a6701_external_capture PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(ppbng_a6701_software_reset PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(ppbng_a6701_recover_idle PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(ppbng_a6701_focus_recorder PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(ppbng_a6701_uno_r4_test_node PRIVATE -Wall -Wextra -Wpedantic)
  endif()
else()
  target_sources(${PROJECT_NAME} PRIVATE src/spinnaker_a6701_backend_stub.cpp)
  target_compile_definitions(${PROJECT_NAME} PUBLIC PPBNG_THERMAL_WITH_SPINNAKER=0)
endif()
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
target_include_directories(${PROJECT_NAME} PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
if(MSVC)
  target_compile_options(${PROJECT_NAME} PRIVATE /W4 /permissive-)
else()
  target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
install(TARGETS ${PROJECT_NAME} EXPORT export_${PROJECT_NAME}
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
if(PPBNG_THERMAL_ENABLE_SPINNAKER)
  install(TARGETS ppbng_spinnaker_inventory ppbng_a6701_readonly_probe ppbng_a6701_health_check ppbng_a6701_acquisition_probe
    ppbng_a6701_external_capture
    ppbng_a6701_software_reset
    ppbng_a6701_recover_idle
    ppbng_a6701_focus_recorder
    ppbng_a6701_uno_r4_test_node
    RUNTIME DESTINATION lib/${PROJECT_NAME})
endif()
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_export_include_directories(include)

find_package(rclcpp REQUIRED)
find_package(std_srvs REQUIRED)
find_package(ppbng_interfaces REQUIRED)
find_package(ppbng_storage REQUIRED)
if(PPBNG_THERMAL_ENABLE_SPINNAKER)
  ament_target_dependencies(ppbng_a6701_uno_r4_test_node rclcpp ppbng_interfaces)
endif()
add_executable(ppbng_thermal_camera_node src/thermal_camera_node.cpp)
target_compile_features(ppbng_thermal_camera_node PRIVATE cxx_std_17)
target_link_libraries(ppbng_thermal_camera_node ${PROJECT_NAME})
ament_target_dependencies(ppbng_thermal_camera_node rclcpp std_srvs ppbng_interfaces ppbng_storage)
install(TARGETS ppbng_thermal_camera_node RUNTIME DESTINATION lib/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_thermal_adapter test/test_thermal_adapter.cpp)
  target_link_libraries(test_thermal_adapter ${PROJECT_NAME})
  if(NOT PPBNG_THERMAL_ENABLE_SPINNAKER)
    ament_add_gtest(test_spinnaker_a6701_backend_stub test/test_spinnaker_a6701_backend_stub.cpp)
    target_link_libraries(test_spinnaker_a6701_backend_stub ${PROJECT_NAME})
  endif()
endif()
ament_package()
